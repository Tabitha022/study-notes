## 迪杰斯特拉求最短路径

>  时间复杂度 ： $\large O((n + m) \ (logn))$
>  空间复杂度 ： $\large O(n + m)$

```cpp
int n, m, s;

    dis[s] = 0;

    q.push({0, s});

        auto x = q.top();
        q.pop();
        int a = -x.first;

            if(dis[v.first] > dis[b] + v.second){//试图更新到 v 点的最短路径

            }
        }
    }
```

## 贝尔曼

**处理含有边权为负数的最短路径**

>  时间复杂度 ： $\large O(nm)$
>  空间复杂度 ： $\large O(n + m)$
>  下面的代码主要为枚举每个点的出边
>  但是如果题目当中  $\large n$ 和 $\large m$ 相差不大，意味着这是一个稀疏图，那么我们可以在 $\large rep(i, 1,  n)$ 里面枚举每条边，时间复杂度 $\large O(nm)$ ，但是实际运行时间会小得多

>  第 $\large n$ 轮还被更新的点，说明达到这个点存在负环

```cpp
vector<pair<int,int>> G[N];//记录有向边和边权
vi dis(n+10，INF);//记录最短路径

dis[1] = 0;

int flag=0;//判断每次循环是否有松弛操作发生，若没有发生，说明所有点的最短路径都已经找到，可以跳出循环。 如果循环结束的时候松弛操作还是发生了，说明存在负环，（也就是这个点的最短路径是负数且存在环，不断地循环重复路径，会让这个最短路径越来越小。
rep(i,1,n){
	flag = 0;
	rep(j, 1, n){
		if(dis[j] == INF)continue;//说明这个点还没有到达过，用这个点去更新没有意义
		
		for(auto v:G[j]){
			if(dis[v.first] > dis[j] + v.second){
				flag = 1;
				dis[v.first] = dis[j] + v.second;
			}
		}
	}
	
	if(flag == 0)break;
}
if(flag == 1) cout<<"存在负环"<<endl;
	
rep(i,1,n)cout<<(dis[i]==INF?-1:dis[i])<<" \n"[i==n];
```

## 区间dp

>  时间复杂度 $\large O(n^3)$
#### 模板题：石子合并
**题意**
有 $\large N$ 堆石子排成一排，其编号为 `1,2,3,...,N` 
每堆石子有一定的质量 $\large m_i$ ，现在要将这 $\large N$ 堆石子合并成为一堆，每次只能合并相邻的两堆，合并的代价为这两堆石子的质量之和，合并后于这两堆石子相邻的石子将与新堆相邻。

合并时由于选择的顺序不同，合并的总代价也不相同。
试找出一种合理的方法，使总的代价最小，并输出最小代价。

$\large 1 \le N \le 300$

**题解**
用 $\large f[l][r]$ 表示合并 $\large l 到 r$ 的最小代价
🐖：$\large f[i][i]$ 表示本来那一堆，代价为 `0`

合并两堆和代价可以表示为：
$\large f[l][r] = f[l][k] +f[k+1][r] +pre[r] -pre[l-1]$

```cpp
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl "\n"
#define rep(i,a,b) for(int i=a;i<=b;i++)
#define per(i,a,b) for(int i=a;i>=b;i--)
#define vi vector<int> 
#define vii vector<vi>
const int INF = 1e18;

void solve(){
    int n;cin>>n;
    vi a(n+10);
    vi pre(n+10);
    vii f(n+10,vi(n+10,INF));

    rep(i,1,n){
        cin>>a[i];
        pre[i]=pre[i-1]+a[i];
        f[i][i]=0;//初始化
    }
    
    rep(len,2,n)//枚举区间长度
        for(int l=1;l+len-1<=n;l++){//枚举左区间
            int r=l+len-1;
            rep(k,l,r-1)//枚举中间k
                f[l][r]=min(f[l][r],f[l][k]+f[k+1][r]+pre[r]-pre[l-1]);
        }

    cout<<f[1][n]<<endl;
    return;
}

signed main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    int T = 1;
    //cin>>T;
    while(T--) solve();
    return 0;
}
```

```cpp
//建立线段树

vector<node> tr(4 * n +1);
vector<int> w(n + 1);

void pushup(int p){
	tr[p].sum = (tr[lc].sum + tr[rc].sum);
	return;
}

void build(int p, int l, int r){
	//先对这个tr[p]进行初始化
	tr[p] = {l, r, 0, 0};
	if(l == r) {
		tr[p].sum = w[l];
		return;
	}
	
	int m = (l + r) >> 1;
	build(lc, l, m);
	build(rc, m+1, r);
	pushup(p);
	return;
}

void solve(){
	rep(i, 1, n) cin >> w[i];
	
	build(1, 1, n);  //分别代表，节点编号，区间左段，区间右端
}
```
**区间内每个点统一加 $\large k$ ——懒标记**
```cpp
//向下传懒标记
void pushdown(int p){
	if(tr[p].add){
		tr[lc].sum += tr[p].add * (tr[lc].r - tr[lc].l + 1);
		tr[rc].sum += tr[p].add * (tr[rc].r - tr[rc].l + 1);
		tr[lc].add += tr[p].add;
		tr[rc].add += tr[p].add;
		tr[p].add = 0;
	}
	return;
}

// p为节点编号 ， x 要加 k 的区间的左端点 ， y 为要加 k 的区间的右端点
void update(int p, int x, int y, int k){
	//如果 p 指向的区间包含在 [x,y] 之内，那么就是 p 的区间内每个点都要加上 k
	if(tr[p].l >= x && tr[p].r <= y){
		tr[p].add += k;
		tr[p].sum += k * (tr[p].r - tr[p].l + 1);
		return;
	}
	pushdown(p);
	int m = (tr[p].l + tr[p].r) >> 1;
	if(x <= m) update(lc , x, y, k);
	if(y >= m + 1) update(rc, x, y, k);
	pushup(p);
}
```

```cpp
//查询区间 ， 总值为 sum
int query(int p, int x, int y){

		return tr[p].sum;
	}

	if(y >= m + 1) query(rc, x, y);
```

## 三分法
适用情况：遇到凸函数求峰点，或者遇到凹函数求谷点的时候
一共存在四个变量，分别为 $\large l, r, midl, midr$
$\large midl = (l + r) >> 1$
$\large midr = (midl + r) >> 1$

示例代码：
```cpp
int l = 0, r = n;
while(l < r){
	int midl = (l + r) >> 1;
	int midr = (midl + r) >> 1;

	if(sum1 < sum2) l = midl;
	else r = midr;
```

## ST表

>  用于维护区间最值，与线段树维护区间最值的核心区别在于， $\large ST$ 表无法中途进行某个值的修改或者某个区间的修改，但在仅仅维护最值这件事上，它得代码量更加少

>  变量解释： 
>  二维数组，$\large st[len][i]$ , 表示 $\large a[i] \ 到 \ a[i + 2^{len} - 1]$ 这个区间的最值
>  

```cpp
vii st(21, vi (n + 1));

rep(i, 1, n) st[0][i] = a[i];

rep(k, 1, 20){ //构造
	for(int j = 1; j + (1 << k) - 1 <= n; j++){
		st[k][j] = max(st[k - 1][j], st[k][j + (1 << (k - 1))]);
	}
}

rep(i, 1, q){ //查询
	int l, r; cin >> l >> r;
	int k = __lg(r - l + 1);
```
